# Import Libraries
import pandas as pd
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

# =========================================
# LOAD DATASETS
# =========================================

adult = pd.read_csv("adult_dataset.csv")
iris = pd.read_csv("iris.csv")

# =========================================
# 1D (LINEAR) DATA VISUALIZATION
# Histogram
# =========================================

plt.figure(figsize=(8,5))
plt.hist(adult['age'], bins=20)
plt.xlabel("Age")
plt.ylabel("Frequency")
plt.title("1D Visualization - Age Distribution")
plt.show()


# =========================================
# 2D (PLANAR) DATA VISUALIZATION
# Scatter Plot
# =========================================

plt.figure(figsize=(8,5))
plt.scatter(iris['SepalLengthCm'], iris['PetalLengthCm'])
plt.xlabel("Sepal Length")
plt.ylabel("Petal Length")
plt.title("2D Visualization - Scatter Plot")
plt.show()


# =========================================
# 3D (VOLUMETRIC) DATA VISUALIZATION
# 3D Scatter Plot
# =========================================

fig = plt.figure(figsize=(8,6))
ax = fig.add_subplot(111, projection='3d')

ax.scatter(
    iris['SepalLengthCm'],
    iris['SepalWidthCm'],
    iris['PetalLengthCm']
)

ax.set_xlabel("Sepal Length")
ax.set_ylabel("Sepal Width")
ax.set_zlabel("Petal Length")

plt.title("3D Visualization - Iris Dataset")
plt.show()
